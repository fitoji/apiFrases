```go
package main

import (
    "aprende-golang/fitoutil"
    "errors"
    "fmt"
    "math"
)

type Persona struct {
    Nombre string
    Edad   int
}
type Rectangulo struct {
    Ancho float64
    Alto  float64
}

func (r Rectangulo) Area() float64 {
    return r.Ancho * r.Alto
}

type MiInt int

const Pi float64 = 3.14

func decirHola() {
    fmt.Println("hola!")
}
func add(x, y int) int {
    return x + y
}
func dividir(x, y int) (int, error) {
    if y == 0 {
        return 0, errors.New("no puedes dividir por cero")
    }
    return x / y, nil
}

func comparar(x, y int) string {
    if x > y {
        return "x es mayor a y"
    }
    return "x es menor o igual a y"
}
func decirAdios() string {
    return "adios!"
}

func main() {
    rect := Rectangulo{
        Alto:  3,
        Ancho: 4,
    }
    fmt.Println(rect.Area())

    p := Persona{
        Nombre: "Pepe",
        Edad:   14,
    }
    fmt.Println(p)
    var otroInt MiInt = 4
    fmt.Println(otroInt)
    despedida := decirAdios()
    defer fmt.Println(despedida)
    // palabra FOR
    for i := 0; i < 5; i++ {
        fmt.Println(i)
    }
    nums := []int{10, 20, 30}
    for _, num := range nums {
        fmt.Println(num)
    }
    var x = 5
    var a, b = 2, 4
    y := 20

    fmt.Println("hola mundo")
    // Manejo de errores para las funciones del paquete fitoutil
    resultadoSuma := fitoutil.Suma(4, 3)
    fmt.Println("Suma fitoutil:", resultadoSuma)

    resultadoResta := fitoutil.Resta(12, 3)
    fmt.Println("Resta fitoutil:", resultadoResta)

    fmt.Println(math.Sqrt(16))
    fmt.Println(x)
    fmt.Println(a, b)
    fmt.Println(y)
    fmt.Println(Pi)
    decirHola()
    suma := add(2, 8)
    fmt.Println(suma)
    division, err := dividir(12, 0)
    if err != nil { // Un linter te obligaría a manejar este error correctamente.
        fmt.Println("hubo un error:", err)
    } else {
        fmt.Println(division)
    }
    comparacion := comparar(4, 1)
    fmt.Println(comparacion)

    day := 4
    switch day {
    case 1:
        fmt.Println("lunes")
    case 2:
        fmt.Println("martes")
    case 3:
        fmt.Println("miercoles")
    default:
        fmt.Println("otro dia")
    }
}
```

```go
    ch1 := make(chan string)
    ch2 := make(chan string)

    go func() { ch1 <- "canal1" }()
    go func() { ch2 <- "canal2" }()

    select {
    case msg1 := <-ch1:
        fmt.Println(msg1)
    case msg2 := <-ch2:
        fmt.Println(msg2)
    }
}

func main() {

	ch := make(chan string)
	go func() {
		time.Sleep(2 * time.Second)
		ch <- "Terminado"
	}()

	select {
	case msg := <-ch:
		fmt.Println(msg)
	case <-time.After(4 * time.Second):
		fmt.Println("sin tiempo")
	}
}
```

## Punteros

```go

func main() {
	x := 10
	p := &x
	*p = 50 // cambiando el valor en memoria en p , desreferenciandolo primero
	fmt.Println(x)
	fmt.Println(p)  //imprime direccion memoria
	fmt.Println(*p) // inprime a donde apunta la variable
}


func incrementarPorValor(x int) {
	x = x + 1
	fmt.Println("dentro de la funcion:", x) //salida 11
}
func incrementarPorPuntero(x *int) {
	*x = *x + 1
	fmt.Println("dentro de funcion de puntero:", *x) //salida 11
}


func main() {
	num := 10
	incrementarPorValor(num)
	fmt.Println("despues de incrementar por valor: ", num)// salida 10, porque imprime num

	incrementarPorPuntero(&num)
	fmt.Println("despues de incrementar por puntero: ", num)//salida 11, porque el puntero modifico el valor de num
}
// Podes manipular variables con la direccion de memoria


type LargeStruct struct {
	Data [1000]int
	Name string
	ID   int
}

func processByValue(ls LargeStruct) {
	fmt.Println("Procesando:", ls.Name)
}

func processByPointer(ls *LargeStruct) {
	fmt.Println("Procesando: ", ls.Name)
}
func main() {
	large := LargeStruct{
		Name: "pepe",
		ID:   1,
	}
	processByValue(large)
	processByPointer(&large)

}
type CuentaDeBanco struct {
	Dueno   string
	Balance float64
}

func (ba CuentaDeBanco) DepositarPorValor(amount float64) {
	ba.Balance += amount
	fmt.Printf("Saldo despues del demosito (metodo por valor): %.2f\n", ba.Balance)
}
func (ba *CuentaDeBanco) DespositarPorPointer(amount float64) {
	ba.Balance += amount
	fmt.Printf("Saldo despues del deposito (metodo por pointer): %.2f \n", ba.Balance)
}
func (ba *CuentaDeBanco) RetirarPorPointer(amount float64) {
	if ba.Balance >= amount {
		ba.Balance -= amount
		fmt.Printf("saldo despues de retirar (por pointer): %.2f \n", ba.Balance)
	} else {
		println("Saldo insuficiente")
	}
}
func main() {
	cuenta := CuentaDeBanco{
		Dueno:   "Fito",
		Balance: 200,
	}
	fmt.Printf("Saldo de %s es de %.2f: \n", cuenta.Dueno, cuenta.Balance)

	//Metodo por valor - no modifica el valor
	cuenta.DepositarPorValor(50)
	fmt.Printf("Saldo real despues de depositar por valor: %.2f \n", cuenta.Balance)
	cuenta.DespositarPorPointer(20)
	fmt.Printf("Saldo despues de depositar por pointer: %.2f \n", cuenta.Balance)
	cuenta.RetirarPorPointer(230)
	fmt.Printf("saldo despues de retirar por pointer %.2f \n", cuenta.Balance)
}
```
